from datetime import datetime

nome = "Dani"
rg = "01234567"
cpf = "000.000.000-00"
ano = "1999"
    
idade = datetime.now().year - ano
    
with open("idade.txt", "w") as arquivo:
    arquivo.write(f"Nome: {nome}\n")
    arquivo.write(f"RG: {rg}\n")
    arquivo.write(f"CPF: {cpf}\n")
    arquivo.write(f"Idade: {idade}\n")
