with open("sem_palavras_repetidas.txt", "w") as arquivo:
  arquivo.write("sol lua terra marte lua terra saturno marte sol venus terra saturno sol venus")

with open("sem_palavras_repetidas.txt", "r") as arquivo:
  palavras = arquivo.read().split()

lista_sem_repeticao = []

for string in palavras:
  if string not in lista_sem_repeticao:
    lista_sem_repeticao.append(string)

print(lista_sem_repeticao)
